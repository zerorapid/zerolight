export function architectProduct(prompt: string): ProductArchetypeSpec {
  const p = prompt.toLowerCase();
  if (/crm|real estate|lead|pipeline|sales|deal/.test(p)) return PRODUCT_ARCHETYPES.crm;
  if (/supply chain|procurement|purchase order|\bpo\b|vendor|supplier/.test(p)) return PRODUCT_ARCHETYPES.procurement;
  if (/agent|copilot|ai assistant|llm|tool call/.test(p)) return PRODUCT_ARCHETYPES["ai-agent-platform"];
  if (/helpdesk|support|ticket|inbox/.test(p)) return PRODUCT_ARCHETYPES.helpdesk;
  if (/ledger|accounting|reconciliation|three[- ]?way|erp|fintech/.test(p)) return PRODUCT_ARCHETYPES.erp;
  return PRODUCT_ARCHETYPES["subscription-saas"];
}
